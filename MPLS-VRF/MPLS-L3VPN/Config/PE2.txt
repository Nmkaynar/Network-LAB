PE2(config)#

conf t

ip vrf Customer-A
 rd 65000:100
 route-target both 100:65000

ip vrf Customer-B
 rd 65100:200
 route-target both 200:65100
 exit



interface Loopback0
 ip address 6.6.6.6 255.255.255.255
 ip ospf 1 area 0
 no sh

inter gi 0/0
 ip vrf forwarding Customer-A
 ip address 192.168.2.1 255.255.255.0
 no sh

inter gi 1/0
 ip vrf forwarding Customer-B
 ip address 192.168.2.1 255.255.255.0
 no sh
 

interface GigabitEthernet2/0
 ip address 10.0.12.2 255.255.255.252
 mpls ip
 ip ospf 1 area 0
 no sh



router ospf 1
 router-id 6.6.6.6
 exit


router bgp 500
 bgp router-id 6.6.6.6
 neighbor 3.3.3.3 remote-as 500
 neighbor 3.3.3.3 update-source loopback0
 neighbor 3.3.3.3 next-hop-self

 address-family vpnv4
  neighbor 3.3.3.3 activate
  neighbor 3.3.3.3 send-community both

 address-family ipv4 vrf Customer-A
  neighbor 192.168.2.254 remote-as 65000
  neighbor 192.168.2.254 activate
  exit

 address-family ipv4 vrf Customer-B
  neighbor 192.168.2.254 remote-as 65100
  neighbor 192.168.2.254 activate
  exit
